#!/bin/bash
# ============================================================
#   VALAKAS BOT - INSTALLER SCRIPT
#   Run as: bash install.valakas
# ============================================================

RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
CYAN='\033[0;36m'
BOLD='\033[1m'
NC='\033[0m'

echo -e "${RED}"
cat << "EOF"
 __   ___   _      _   _  __   _   ___  
 \ \ / / \ | |    /_\ | |/ /  /_\ / __|
  \ V /| .`| |   / _ \| ' <  / _ \\__ \
   \_/ |_|\_|___/_/ \_\_|\_\/_/ \_\___/
             VPS BOT INSTALLER
EOF
echo -e "${NC}"

echo -e "${CYAN}[1/5] Updating system...${NC}"
apt update -y && apt upgrade -y

echo -e "${CYAN}[2/5] Installing Python & pip...${NC}"
apt install -y python3 python3-pip curl screen git

echo -e "${CYAN}[3/5] Installing bot requirements...${NC}"
pip3 install discord.py psutil aiohttp

echo -e "${CYAN}[4/5] Setting up Valakas bot...${NC}"
mkdir -p /root/valakas
cp valakas.py /root/valakas/

echo -e "${YELLOW}[!] Enter your Discord Bot Token:${NC}"
read -r TOKEN

# Replace token in file
sed -i "s/YOUR_BOT_TOKEN_HERE/$TOKEN/" /root/valakas/valakas.py

echo -e "${YELLOW}[!] Enter your Discord User ID (for owner commands):${NC}"
read -r OWNER_ID

sed -i "s/123456789/$OWNER_ID/" /root/valakas/valakas.py

echo -e "${CYAN}[5/5] Starting Valakas Bot in screen session...${NC}"
screen -dmS valakas bash -c "cd /root/valakas && python3 valakas.py"

echo -e "${GREEN}"
echo "══════════════════════════════════════"
echo "  ✅  VALAKAS BOT IS NOW RUNNING!"
echo "══════════════════════════════════════"
echo "  📺 View bot:    screen -r valakas"
echo "  🛑 Stop bot:    screen -X -S valakas quit"
echo "  🔄 Restart:     !restart in Discord"
echo "══════════════════════════════════════"
echo -e "${NC}"
